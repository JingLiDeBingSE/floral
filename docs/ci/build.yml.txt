name: build

on:
  push:
    branches: [ main, master ]
  pull_request:
  workflow_dispatch:

jobs:
  windows:
    runs-on: windows-latest

    steps:
      - uses: actions/checkout@v4

      # 便携版需要内嵌 sidecars.zip（构建产物，不在仓库里），CI 只验证能编过
      - name: Configure (MSVC)
        run: cmake -S . -B build -G "Visual Studio 17 2022" -DFLORAL_PORTABLE=ON -DFLORAL_EMBED_SERVICES=OFF

      - name: Build
        run: cmake --build build --config Release

      - name: Show output
        run: |
          Get-ChildItem build -Filter *.exe | ForEach-Object {
            "{0}  {1:N1} MB" -f $_.Name, ($_.Length / 1MB)
          }

      - uses: actions/upload-artifact@v4
        with:
          name: floral-msvc
          path: build/*.exe
          if-no-files-found: error
